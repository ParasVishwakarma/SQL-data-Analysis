CREATE DATABASE SalesAnalysisDB;

USE SalesAnalysisDB;

CREATE TABLE sales_data (
    sale_id INT PRIMARY KEY AUTO_INCREMENT,
    product_id INT NOT NULL,
    category VARCHAR(100) NOT NULL,
    sale_date DATE NOT NULL,
    amount DECIMAL(10, 2) NOT NULL
);

CREATE TABLE products (
    product_id INT PRIMARY KEY,
    product_name VARCHAR(255) NOT NULL,
    price DECIMAL(10, 2) NOT NULL
);

-- Step 1: Create a CTE to calculate total sales per product
WITH product_sales AS (
    SELECT
        sd.product_id,
        p.product_name,
        sd.category,
        SUM(sd.amount) AS total_sales
    FROM
        sales_data sd
    JOIN
        products p
    ON
        sd.product_id = p.product_id
    GROUP BY
        sd.product_id, p.product_name, sd.category
),

-- Step 2: Add window functions for ranking and cumulative totals
ranked_sales AS (
    SELECT
        ps.category,
        ps.product_id,
        ps.product_name,
        ps.total_sales,
        RANK() OVER (PARTITION BY ps.category ORDER BY ps.total_sales DESC) AS rank_within_category,
        SUM(ps.total_sales) OVER (PARTITION BY ps.category ORDER BY ps.total_sales DESC) AS cumulative_sales
    FROM
        product_sales ps
),

-- Step 3: Calculate overall metrics for each category
category_metrics AS (
    SELECT
        category,
        AVG(total_sales) AS avg_sales_per_category,
        MAX(total_sales) AS max_sales_per_category
    FROM
        product_sales
    GROUP BY
        category
)

-- Step 4: Combine all data for final output
SELECT
    rs.category,
    rs.product_id,
    rs.product_name,
    rs.total_sales,
    rs.rank_within_category,
    rs.cumulative_sales,
    cm.avg_sales_per_category,
    cm.max_sales_per_category
FROM
    ranked_sales rs
JOIN
    category_metrics cm
ON
    rs.category = cm.category
ORDER BY
    rs.category, rs.rank_within_category;
